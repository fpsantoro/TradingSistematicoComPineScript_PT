//@version=5
strategy("Trendline", 
     overlay = false, 
     initial_capital = 100000, //default_qty_type = strategy.cash, default_qty_value = 10000,
     commission_type = strategy.commission.percent, 
     commission_value = 0.04, 
     pyramiding = 0)

// ═══════════════════════════════════════════════════════════════
// INPUTS
// ═══════════════════════════════════════════════════════════════
group_session = "Configuração da Sessão"
session_duration_hours = input.float(7.0, "Duração da Sessão (horas)", minval=0.1, step=0.1, group=group_session,
 tooltip="Informe a duração da sessão de negociação em horas (ex: 7.0h para a B3")

grupo_estrategia    = "Parâmetros da Estratégia"
operar_long         = input.bool(true, "Operar Long (true) ou Short (false)", group = grupo_estrategia)
razao_risco         = input.float(1.2, "Razão Risco/Recompensa mínima", step = 0.1, group = grupo_estrategia)
capital_risco       = input.float(500, "Capital arriscado por operação", step=50, group=grupo_estrategia,
 tooltip="Valor monetário máximo que está disposto a perder em cada operação (ex: 1000 = 1% de 100.000)")

grupo_sltp       = "Stop-Loss e Take-Profit"
beta             = input.float(1.0, "ATR multiplier", minval = 0.1, step = 0.1, group = grupo_sltp)
stop_loss_perc   = input.float(2.0, "Stop-Loss (%)",
 minval = 0.1, step = 0.1, group = grupo_sltp) / 100

grupo_swings     = "Níveis de Swing"
mostrar_swings   = input.bool(true, "Mostrar Swing High/Low Diário", group = grupo_swings)
cor_resistencia  = color.new(color.teal, 0)
cor_suporte      = color.new(color.maroon, 0)

grupo_filtro     = "Indicador Elder-ray"
ER_period = input.int(13, "Period for the MA which the Bull/Bear Power indicator is calculated", minval=1, group=grupo_filtro)
smoothingER = input.string("EMA", "MA Smoothing Method", options=["RMA", "SMA", "EMA", "WMA"], group=grupo_filtro)

group_visual   = "Visual"
showTrendLines = input.bool(true, "Show Resistance Trendlines", group=group_visual)
maxLines       = input.int(1, "Maximum number of independent lines", minval=1, maxval=20, group=group_visual)

// ─────────────────────────────────────────────────────────────
// Dados diários
// ─────────────────────────────────────────────────────────────
[maxima_d, minima_d] = request.security(syminfo.tickerid, "D",
 [high, low], lookahead = barmerge.lookahead_off)
[maxima_d1, minima_d1] = request.security(syminfo.tickerid, "D",
 [high[1], low[1]], lookahead = barmerge.lookahead_off)
[maxima_d2, minima_d2] = request.security(syminfo.tickerid, "D",
 [high[2], low[2]], lookahead = barmerge.lookahead_off)

atrD = request.security(syminfo.tickerid, "D", ta.atr(14), lookahead = barmerge.lookahead_off)


var float ultimo_swing_high = na
var float penultimo_swing_high = na
var float ultimo_swing_low  = na
var float penultimo_swing_low  = na
var int lastSwingBar = na
var int prevSwingBar = na

var float preco_entrada     = na
var float nivel_stop        = na
var float nivel_target      = na
var bool  em_posicao        = false

// Cálculo do número de candles por dia
timeframe_seconds = timeframe.in_seconds(timeframe.period)
session_seconds   = session_duration_hours * 3600
candles_per_day   = math.ceil(session_seconds / timeframe_seconds)

novo_dia = ta.change(time("D")) != 0

swing_high_diario = novo_dia and not na(maxima_d)
 and not na(maxima_d1) and not na(maxima_d2)
 and maxima_d1 > maxima_d and maxima_d1 > maxima_d2
swing_low_diario  = novo_dia and not na(minima_d)
 and not na(minima_d1) and not na(minima_d2)
 and minima_d1 < minima_d and minima_d1 < minima_d2

if swing_high_diario
    penultimo_swing_high := ultimo_swing_high
    ultimo_swing_high    := maxima_d1

    prevSwingBar  := lastSwingBar 
    lastSwingBar  := bar_index - 2 * candles_per_day

if swing_low_diario
    penultimo_swing_low := ultimo_swing_low
    ultimo_swing_low    := minima_d1

plot(mostrar_swings ? ultimo_swing_high : na,
 "Resistência Diária", color = cor_resistencia,
 style = plot.style_circles, linewidth = 1, offset = -candles_per_day,
 force_overlay = true)
plot(mostrar_swings ? ultimo_swing_low  : na,
 "Suporte Diário", color = cor_suporte,
 style =plot.style_circles, linewidth = 1, offset = -candles_per_day,
 force_overlay = true)

// ═══════════════════════════════════════════════════════════════
// LINEE INDIPENDENTI
// ═══════════════════════════════════════════════════════════════
var line[] trendLines = array.new_line()

if swing_high_diario and not na(penultimo_swing_high) and not na(ultimo_swing_high)
    if penultimo_swing_high > ultimo_swing_high and prevSwingBar < lastSwingBar
        line newLine = line.new(prevSwingBar, penultimo_swing_high, lastSwingBar, ultimo_swing_high,
             xloc   = xloc.bar_index,
             extend = extend.right,
             color  = showTrendLines ? color.orange : color.new(color.orange, 100),
             width  = 1,
             style  = line.style_solid, force_overlay=true)
        
        array.push(trendLines, newLine)
        
        if array.size(trendLines) > maxLines
            line oldLine = array.shift(trendLines)
            line.delete(oldLine)

// ═══════════════════════════════════════════════════════════════
// VALORE DELLA TRENDLINE PIÙ RECENTE (per i segnali)
// ═══════════════════════════════════════════════════════════════
var float currentTLValue = na

if array.size(trendLines) > 0
    line lastLine = array.get(trendLines, array.size(trendLines) - 1)
    
    int   x1 = line.get_x1(lastLine)
    float y1 = line.get_y1(lastLine)
    int   x2 = line.get_x2(lastLine)
    float y2 = line.get_y2(lastLine)
    
    if x2 != x1
        float slope = (y2 - y1) / (x2 - x1)
        currentTLValue := y1 + slope * (bar_index - x1)
    else
        currentTLValue := y1
else
    currentTLValue := na

// ═══════════════════════════════════════════════════════════════
// INDICATORI
// ═══════════════════════════════════════════════════════════════

// Custom Moving Average function (supports multiple smoothing types)
ma_function(source, length, smoothing) =>
    switch smoothing
        "RMA" => ta.rma(source, length)
        "SMA" => ta.sma(source, length)
        "EMA" => ta.ema(source, length)
        => ta.wma(source, length)

ERma = ma_function(close, ER_period * candles_per_day, smoothingER)

stop_loss = math.sqrt(1+candles_per_day) * ma_function(ta.tr(true), 14 * candles_per_day, "RMA")

BullPwr = maxima_d - ERma
BearPwr = minima_d - ERma

plot(BullPwr[1], "Bull Power", color=color.new(color.green, 25), style=plot.style_columns)//, offset=-candles_per_day)
plot(BearPwr[1], "Bear Power", color=color.new(color.red, 50), style=plot.style_columns)//, offset=-candles_per_day)

hline(0, "Zero line", color.black, hline.style_dotted)

// ═══════════════════════════════════════════════════════════════
// VARIABILI DI GESTIONE POSIZIONE
// ═══════════════════════════════════════════════════════════════
var bool  in_position       = false
var float entry_price       = na
var float stop_loss_level   = na
var float take_profit_level = na
var float sizing            = na

// ═══════════════════════════════════════════════════════════════
// CONDIZIONE DI INGRESSO LONG
// ═══════════════════════════════════════════════════════════════
time_condition = timestamp(year(time), month(time), dayofmonth(time)) >= timestamp(2022, 01, 01)

// Core conditions
bullish_trend    = BearPwr < 0 and close > ERma

bool longCondition = (not in_position and not na(currentTLValue) and ta.crossover(close, currentTLValue)
 and time_condition and bullish_trend)

// Rileva uscita per stop loss o take profit
exit_by_strategy_exit = in_position and strategy.position_size == 0

if longCondition
    in_position     := true
    entry_price     := close
    stop_loss_level := math.min(entry_price - beta * atrD, entry_price * (1 - stop_loss_perc))
    risco_por_unidade = math.abs(stop_loss_level - entry_price)
    take_profit_level := entry_price + risco_por_unidade * razao_risco

    qty = capital_risco / risco_por_unidade
    strategy.entry("Long", strategy.long, qty=qty)
    strategy.exit("TP/SL", "Long", stop=stop_loss_level, limit=take_profit_level)

// Reset quando la posizione si chiude
if exit_by_strategy_exit
    strategy.close("Long", qty_percent=100, immediately=true)  // Chiude la posizione se non già chiusa
    in_position       := false
    entry_price       := na
    stop_loss_level   := na
    take_profit_level := na
    sizing            := na

// ═══════════════════════════════════════════════════════════════
// VISUALIZZAZIONE SL / TP (stile della strategia di riferimento)
// ═══════════════════════════════════════════════════════════════
plot(in_position ? take_profit_level : na, "Take Profit", color=color.purple, style=plot.style_linebr, linewidth=1, force_overlay=true)
plot(in_position ? stop_loss_level   : na, "Stop Loss",   color=color.black,  style=plot.style_linebr, linewidth=1, force_overlay=true)

bgcolor(bool(ta.change(time("W"))) ? color.new(color.silver, 90) : na, title="New Week")
bgcolor(bool(ta.change(time("W"))) ? color.new(color.silver, 90) : na, title="New Week", force_overlay=true)
